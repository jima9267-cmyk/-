const SHEET_ID = '1G7V95ovfAJ_WJauhv_zuaNB4nejg194qTyZSOAroa4Y'; const SHEET_NAME = 'submissions'; function sha256Hex(text) { const rawHash = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, text, Utilities.Charset.UTF_8 ); return rawHash .map(byte => { const v = (byte + 256) % 256; return v.toString(16).padStart(2, '0'); }) .join(''); } function getSheet() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['timestamp', 'student_key', 'name', 'student_id', 'pin_hash', 'intent', 'scene']); } return sheet; } function doPost(e) { try { const body = JSON.parse(e.postData.contents || '{}'); const action = body.action || 'save'; const name = String(body.name || '').trim(); const studentId = String(body.studentId || '').trim(); const pin = String(body.pin || '').trim(); if (!studentId || !pin || pin.length < 4) { return jsonResponse({ ok: false, message: '학번과 4자리 이상 확인 번호(PIN)를 입력해 주세요.' }); } const studentKey = sha256Hex(`sid:${studentId}`); const pinHash = sha256Hex(`${studentId}:${pin}`); const sheet = getSheet(); const lastRow = sheet.getLastRow(); if (action === 'load') { if (lastRow < 2) { return jsonResponse({ ok: false, message: '저장된 기록을 찾을 수 없습니다.' }); } const rows = sheet.getRange(2, 1, lastRow - 1, 7).getValues(); // 가장 최근 제출 건부터 역순 탐색 for (let i = rows.length - 1; i >= 0; i--) { const [ts, rowKey, rowName, rowSid, rowPinHash, intent, scene] = rows[i]; const idMatch = String(rowSid).trim() === studentId || String(rowKey) === studentKey; const nameMatch = !name || String(rowName).trim() === name; if (idMatch && nameMatch) { if (String(rowPinHash) !== pinHash) { return jsonResponse({ ok: false, message: '확인 번호(PIN)가 일치하지 않습니다.' }); } return jsonResponse({ ok: true, message: '본인 확인이 완료되었습니다.', data: { updatedAt: ts, intent: String(intent || ''), scene: String(scene || '') } }); } } return jsonResponse({ ok: false, message: '일치하는 제출 기록이 없습니다.' }); } // action === 'save' if (!name) { return jsonResponse({ ok: false, message: '이름을 입력해 주세요.' }); } const intent = String(body.intent || '').trim(); const scene = String(body.scene || '').trim(); if (!intent || !scene) { return jsonResponse({ ok: false, message: '창작 의도와 장면 묘사를 모두 입력해 주세요.' }); } // 기존 동일 학번의 PIN이 있다면 일치 여부 확인 if (lastRow >= 2) { const rows = sheet.getRange(2, 1, lastRow - 1, 7).getValues(); for (let i = rows.length - 1; i >= 0; i--) { const [, , , rowSid, rowPinHash] = rows[i]; if (String(rowSid).trim() === studentId && String(rowPinHash) && String(rowPinHash) !== pinHash) { return jsonResponse({ ok: false, message: '이미 등록된 학번의 확인 번호(PIN)와 다릅니다.' }); } } } const now = new Date(); sheet.appendRow([now, studentKey, name, studentId, pinHash, intent, scene]); return jsonResponse({ ok: true, message: '데이터베이스에 안전하게 저장되었습니다.', data: { updatedAt: now.toISOString() } }); } catch (err) { return jsonResponse({ ok: false, message: '서버 처리 중 오류가 발생했습니다: ' + err.message }); } } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
